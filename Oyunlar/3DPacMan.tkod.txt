// ===== SAHNE ÇİZ =====
Fonksiyon SahneCiz()
  Kanvas3DTemizle(kv);
  
  // Duvarlar - parlak mavi
  Sayi x = 0;
  Tekrarla HAR_GEN Kez
    Sayi y = 0;
    Tekrarla HAR_YUK Kez
      Eger harita[y][x] == 1 Ise
        Kanvas3DKup(kv, x, 0.5, y, "#3355cc", 0.5);
      EgerBitti
      // Noktalar - beyaz parlak
      Eger harita[y][x] == 2 Ise
        Kanvas3DKure(kv, x, 0.2, y, "#ffffff", 0.07);
      EgerBitti
      y = y + 1;
    TekrarlaBitti
    x = x + 1;
  TekrarlaBitti
  
  // Pac-Man - BÜYÜK SARI (BURADA!)
  Eger oyunAktif Ise
    Kanvas3DKure(kv, pacX, 0.45, pacY, "#ffdd00", 0.45);
  EgerBitti
  
  // Hayaletler - küçük
  Sayi h = 0;
  Tekrarla 4 Kez
    Eger oyunAktif Ise
      Kanvas3DKure(kv, hayaletler[h][0], 0.4, hayaletler[h][1], hayRenk[h], 0.25);
    EgerBitti
    h = h + 1;
  TekrarlaBitti
  
  Kanvas3DKamera(kv, 4.5, 10, 14, 4.5, 0, 6);
  Kanvas3DIsik(kv, 5, 12, 8, 0.5);
  Kanvas3DRender(kv);
FonksiyonBitti

// ===== HAYALET AI =====
Fonksiyon HayaletHareket()
  Eger tickSay % 5 != 0 Ise Don; EgerBitti // 5 tickte bir hareket = yavaş!
  
  Sayi h = 0;
  Tekrarla 4 Kez
    Sayi hx = hayaletler[h][0];
    Sayi hy = hayaletler[h][1];
    Sayi yon = hayYon[h];
    
    Sayi rast = Rastgele(0, 8);
    Eger rast < 3 Ise
      yon = Rastgele(0, 3);
    EgerBitti
    
    Sayi nx = hx; Sayi ny = hy;
    Eger yon == 0 Ise ny = hy - 1; EgerBitti
    Eger yon == 1 Ise nx = hx + 1; EgerBitti
    Eger yon == 2 Ise ny = hy + 1; EgerBitti
    Eger yon == 3 Ise nx = hx - 1; EgerBitti
    
    Eger nx >= 0 Ve nx < HAR_GEN Ve ny >= 0 Ve ny < HAR_YUK Ise
      Eger harita[ny][nx] != 1 Ise
        hayaletler[h][0] = nx;
        hayaletler[h][1] = ny;
        hayYon[h] = yon;
      Degilse
        hayYon[h] = Rastgele(0, 3);
      EgerBitti
    EgerBitti
    
    // Pac-Man yakalandı mı?
    Eger hayaletler[h][0] == pacX Ve hayaletler[h][1] == pacY Ise
      oyunAktif = Yanlis;
      zm.Durdur();
      SesEfekti(ses, "alarm");
      lblDurum.yazi = hayIsim[h] + " hayalet yakaladı!";
      Mesaj("💀 " + hayIsim[h] + " hayalet yakaladı!\nSkor: " + MetineKonvert(skor));
    EgerBitti
    
    h = h + 1;
  TekrarlaBitti
FonksiyonBitti

// ===== OYUN TICK =====
Fonksiyon OyunTick()
  Eger oyunAktif == Yanlis Ise Don; EgerBitti
  
  tickSay = tickSay + 1;
  
  // Pac-Man hareket
  Sayi nx = pacX; Sayi ny = pacY;
  Eger pacYon == 0 Ise ny = pacY - 1; EgerBitti
  Eger pacYon == 1 Ise nx = pacX + 1; EgerBitti
  Eger pacYon == 2 Ise ny = pacY + 1; EgerBitti
  Eger pacYon == 3 Ise nx = pacX - 1; EgerBitti
  
  Eger nx >= 0 Ve nx < HAR_GEN Ve ny >= 0 Ve ny < HAR_YUK Ise
    Eger harita[ny][nx] != 1 Ise
      pacX = nx;
      pacY = ny;
      Eger harita[pacY][pacX] == 2 Ise
        harita[pacY][pacX] = 0;
        skor = skor + 10;
        lblSkor.yazi = "SKOR: " + MetineKonvert(skor);
        SesEfekti(ses, "bip");
      EgerBitti
    EgerBitti
  EgerBitti
  
  HayaletHareket();
  SahneCiz();
FonksiyonBitti
